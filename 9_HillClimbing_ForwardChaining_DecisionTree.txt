
def f(x):
    return -(x**2)+10*x

def hill_climb(start):
    current=start
    while True:
        if f(current+1)>f(current):
            current+=1
        elif f(current-1)>f(current):
            current-=1
        else:
            return current,f(current)

x,value=hill_climb(2)
print("Maximum at x =",x)
print("Maximum value =",value)

facts={"furry","barks"}
rules=[
    ({"furry","barks"},"is_dog"),
    ({"is_dog"},"has_tail"),
    ({"has_tail"},"wags_tail"),
    ({"furry","meows"},"is_cat")
]
print("Initial facts:",facts)
changed=True
while changed:
    changed=False
    for conditions,conclusion in rules:
        if conditions.issubset(facts) and conclusion not in facts:
            facts.add(conclusion)
            changed=True
            print("Derived:",conclusion,"from",conditions)
print("Final facts:",facts)

import numpy as np
from sklearn.tree import DecisionTreeClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score

data=np.array([
    [6,148,72,35,0,33.6,0.627,50,1],
    [1,85,66,29,0,26.6,0.351,31,0],
    [8,183,64,0,0,23.3,0.672,32,1],
    [1,89,66,23,94,28.1,0.167,21,0],
    [0,137,40,35,168,43.1,2.288,33,1],
    [5,116,74,0,0,25.6,0.201,30,0],
    [3,78,50,32,88,31,0.248,26,1],
    [10,115,0,0,0,35.3,0.134,29,0],
    [2,197,70,45,543,30.5,0.158,53,1],
    [8,125,96,0,0,37.6,0.232,54,1],
    [4,110,92,0,0,37.6,0.191,30,0]
])
X=data[:,:-1]
y=data[:,-1]
X_train,X_test,y_train,y_test=train_test_split(X,y,test_size=0.30,random_state=42)
print("Decision Tree Classifier")
print("Training samples:",len(X_train))
print("Testing samples:",len(X_test))
best_accuracy=0
best_model=None
for criterion in ['gini','entropy']:
    for depth in [2,3,4]:
        model=DecisionTreeClassifier(criterion=criterion,max_depth=depth,random_state=42)
        model.fit(X_train,y_train)
        y_pred=model.predict(X_test)
        accuracy=accuracy_score(y_test,y_pred)
        print("Criterion:",criterion,"Max Depth:",depth,"Accuracy:",accuracy)
        if accuracy>best_accuracy:
            best_accuracy=accuracy
            best_model=model
print("\nBest Model")
print("Criterion:",best_model.criterion)
print("Max Depth:",best_model.max_depth)
print("Best Accuracy:",best_accuracy)
