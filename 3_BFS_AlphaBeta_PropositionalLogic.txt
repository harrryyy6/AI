from collections import deque
def bfs_traversal(graph,start):
    visited=set()
    queue=deque([start])
    visited.add(start)
    traversal_order=[]
    while queue:
        node=queue.popleft()
        traversal_order.append(node)
        for neighbor in graph[node]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)
    return traversal_order
graph={
    1:[2,3],
    2:[1,4,5],

print("BFS Traversal starting from",start_node,":",result)

def alpha_beta(node,depth,is_max,alpha,beta,leaves):
    if depth==0:
        return leaves[node]
    if is_max:
        best=float("-inf")
        for i in range(2):
            val=alpha_beta(node*2+i,depth-1,False,alpha,beta,leaves)
            best=max(best,val)
            alpha=max(alpha,best)
            if beta<=alpha:
                break
        return best
    else:
        best=float("inf")
        for i in range(2):
            val=alpha_beta(node*2+i,depth-1,True,alpha,beta,leaves)
            best=min(best,val)
            beta=min(beta,best)
            if beta<=alpha:
                break
        return best
result=alpha_beta(0,2,True,float("-inf"),float("inf"),values)
print("Alpha-Beta result:",result)
print("Leaf values:",values)
