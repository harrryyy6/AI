from collections import deque
def water_jug():
    visited=set()
    queue=deque([(0,0)])
    while queue:
        x,y=queue.popleft()
        if (x,y) in visited:
            continue
        visited.add((x,y))
        if x==2 or y==2:
            print("Solution:",(x,y))
            return
        states=[
            (4,y),
            (x,3),
            (0,y),
            (x,0),
            (max(0,x-(3-y)),min(3,x+y)),
            (min(4,x+y),max(0,y-(4-x)))
        ]
        for state in states:
            queue.append(state)
water_jug()

def is_safe(v,graph,color,c):
    for i in range(len(graph)):
        if graph[v][i]==1 and color[i]==c:
            return False
    return True

def graph_coloring(graph,m,color,v):
    if v==len(graph):
        return True
    for c in range(1,m+1):
        if is_safe(v,graph,color,c):
            color[v]=c
            if graph_coloring(graph,m,color,v+1):
                return True
            color[v]=0
    return False

graph=[
    [0,1,1,1,1],
    [1,0,1,1,1],
    [1,1,0,1,1],
    [1,1,1,0,1],
    [1,1,1,1,0]
]
m=5
vertices=['1','2','3','4','5']
color=[0]*len(graph)
print("\n---Backtracking Algorithm for CSP---")
print("Vertices:",vertices)
print("Number of Colors:",m)
print()
if graph_coloring(graph,m,color,0):
    print("Solution Found:")
    for i in range(len(vertices)):
        print(vertices[i],"-> Color",color[i])
else:
    print("No solution exists.")

tree={
    "root":{"kids":["A","B"],"type":"max"},
    "A":{"kids":["A1","A2"],"type":"min"},
    "B":{"kids":["B1","B2"],"type":"min"},
    "A1":{"val":3},
    "A2":{"val":7},
    "B1":{"val":2},
    "B2":{"val":9}
}
def evaluate(name):
    node=tree[name]
    if "val" in node:
        return node["val"]
    scores=[]
    for child in node["kids"]:
        scores.append(evaluate(child))
    if node["type"]=="max":
        return max(scores)
    return min(scores)
print("Game Tree Structure:")
for name,node in tree.items():
    if "kids" in node:
        print(f" {name} ({node['type']}): {node['kids']}")
    else:
        print(f" {name} (value): {node['val']}")
print("Minimax value from root:",evaluate("root"))
