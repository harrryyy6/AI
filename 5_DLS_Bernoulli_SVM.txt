def dls(graph,node,goal,limit,path=[]):
    path=path+[node]
    if node==goal:
        return path
    if limit<=0:
        return None
    for child in graph.get(node,[]):
        result=dls(graph,child,goal,limit-1,path)
        if result:
            return result
    return None
graph={
    'A':['B','C'],
    'B':['D','E'],
    'C':['F'],
    'D':[],
    'E':[],
    'F':[]
}
start='A'
goal='F'
limit=2
result=dls(graph,start,goal,limit)
print("Start Node:",start)
print("Goal Node:",goal)
print("Depth Limit:",limit)
print("Path found:",result)

import numpy as np
from sklearn.datasets import make_classification
from sklearn.naive_bayes import BernoulliNB
X,y=make_classification(n_samples=50,n_features=2,n_informative=2,n_redundant=0,n_classes=2,random_state=1)
model=BernoulliNB()
model.fit(X,y)
test_values=np.array([[0,0],[0,1],[1,0],[1,1]])
prediction=model.predict(test_values)
print("Bernoulli Naive Bayes Classifier")
print("Number of samples:",50)
print("Number of classes:",2)
print("\nInput Values:")
print(test_values)
print("\nOutput:")
print(prediction)
