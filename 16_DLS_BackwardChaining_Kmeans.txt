
def dls(graph,node,goal,limit,path=None):
    if path is None:
        path=[]
    path=path+[node]
    if node==goal:
        return path
    if limit<=0:
        return None
    for child in graph.get(node,[]):
        result=dls(graph,child,goal,limit-1,path)
        if result:
            return result
    return None

graph={1:[2,3],2:[4,5],3:[6],4:[],5:[],6:[]}

goal=6
limit=2

print("Start Node:",start)

print("Depth Limit:",limit)
print("Path found:",result)

facts={"furry","meows"}
rules=[
    (["furry","meows"],"is_cat"),
    (["is_cat"],"likes_milk"),
    (["likes_milk"],"drinks_milk")
]

def bc(goal,known,depth=0):
    indent=" "*depth
    if goal in known:
        print(indent+goal+" is a known fact")
        return True
    for premises,conclusion in rules:
        if conclusion==goal:
            print(indent+"Trying rule:",premises,"->",conclusion)
            if all(bc(p,known,depth+1) for p in premises):
                known.add(goal)
                print(indent+conclusion+" PROVED")
                return True
    print(indent+goal+" cannot be proved")
    return False

for goal in ["is_cat","drinks_milk","is_dog"]:
    print("\nProve:",goal)
    result=bc(goal,facts)
    print("Result:",result)

import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

data=pd.read_csv("country_data.csv")
print("Dataset:")
print(data.head())

features=['child_mortality','exports','health','imports','income','inflation','life_expectancy','total_fertility']
X=data[features]

print("\nMissing Values:")
print(X.isnull().sum())

scaler=StandardScaler()
X_scaled=scaler.fit_transform(X)

wcss=[]
for k in range(3,11):
    model=KMeans(n_clusters=k,random_state=42,n_init=10)
    model.fit(X_scaled)
    wcss.append(model.inertia_)
    print("K =",k,"WCSS =",round(model.inertia_,2))

plt.figure(figsize=(8,5))
plt.plot(range(3,11),wcss,marker='o')
plt.xlabel("Number of Clusters (K)")
plt.ylabel("WCSS")
plt.title("Elbow Method for K-Means Clustering")
plt.grid(True)
plt.show()
