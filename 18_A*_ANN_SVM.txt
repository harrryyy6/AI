
import heapq

graph={
    'A':[('B',1),('C',4)],
    'B':[('D',2),('E',5)],
    'C':[('E',1)],
    'D':[('F',3)],
    'E':[('F',2)],
    'F':[]
}

heuristic={
    'A':6,
    'B':5,
    'C':3,
    'D':3,
    'E':2,
    'F':0
}

def a_star_search(start,goal):
    open_list=[(heuristic[start],0,start,[start])]
    visited=set()

    while open_list:
        f,g,node,path=heapq.heappop(open_list)

        if node==goal:
            return path,g

        if node in visited:
            continue

        visited.add(node)

        for neighbor,cost in graph[node]:
            if neighbor not in visited:
                new_g=g+cost
                new_f=new_g+heuristic[neighbor]
                heapq.heappush(open_list,(new_f,new_g,neighbor,path+[neighbor]))

    return None,None

start='A'
goal='F'

path,cost=a_star_search(start,goal)

print("A* Search Path:",path)
print("Total Cost:",cost)

import numpy as np

import numpy as np
import matplotlib.pyplot as plt
from sklearn import svm
from sklearn.datasets import make_moons

X, y = make_moons(n_samples=200, noise=0.2, random_state=42)

kernels = ['linear', 'rbf', 'poly', 'sigmoid']
knames = ['Linear', 'RBF', 'Polynomial', 'Sigmoid']

x_min = X[:, 0].min() - 0.5
x_max = X[:, 0].max() + 0.5
y_min = X[:, 1].min() - 0.5
y_max = X[:, 1].max() + 0.5

xx, yy = np.meshgrid(
    np.arange(x_min, x_max, 0.05),
    np.arange(y_min, y_max, 0.05)
)

fig, axes = plt.subplots(2, 2, figsize=(9, 9))

for axis, kernel, name in zip(axes.ravel(), kernels, knames):
    clf = svm.SVC(kernel=kernel, C=1.0, gamma='scale')
    clf.fit(X, y)

    Z = clf.predict(np.c_[xx.ravel(), yy.ravel()])
    Z = Z.reshape(xx.shape)

    axis.contourf(xx, yy, Z, alpha=0.4, cmap='coolwarm')
    axis.scatter(X[:, 0], X[:, 1], c=y, cmap='coolwarm', edgecolor='k', s=25)
    axis.set_title('Decision Boundary - ' + name + ' Kernel')
    axis.set_xlabel('Feature 1')
    axis.set_ylabel('Feature 2')

plt.suptitle('SVM Decision Boundaries for Different Kernels')
plt.tight_layout()
plt.show()
